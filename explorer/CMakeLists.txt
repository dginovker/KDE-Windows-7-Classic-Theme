cmake_minimum_required(VERSION 3.20)
project(win7-explorer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Widgets PrintSupport)
find_package(KF6Config REQUIRED)
find_package(KF6GuiAddons REQUIRED)
find_package(KF6IconThemes REQUIRED)
find_package(KF6Service REQUIRED)
find_package(KF6KIO 6.17 REQUIRED)
set(CMAKE_AUTORCC ON)
if(CMAKE_INSTALL_PREFIX STREQUAL "/usr")
    add_subdirectory(../plasma/desktop-menu desktop-menu)
endif()
add_subdirectory(../classic classic)
add_executable(win7-explorer main.cpp FolderProperties.cpp assets/assets.qrc)
target_link_libraries(win7-explorer classic-raster Qt6::Widgets Qt6::PrintSupport KF6::ConfigCore KF6::GuiAddons KF6::IconThemes KF6::Service KF6::KIOFileWidgets KF6::KIOWidgets KF6::KIOGui)
include(GNUInstallDirs)
install(TARGETS win7-explorer DESTINATION ${CMAKE_INSTALL_BINDIR})
# Absolute Exec path: a per-user install's ~/.local/bin is not on plasmashell's PATH.
configure_file(org.win7classic.explorer.desktop.in org.win7classic.explorer.desktop @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/org.win7classic.explorer.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
foreach(size 16 32 48 256)
    install(FILES assets/explorer-${size}.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/${size}x${size}/apps RENAME org.win7classic.explorer.png)
endforeach()
