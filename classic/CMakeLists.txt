cmake_minimum_required(VERSION 3.20)
project(win7-classic-primitives LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Quick Qml WaylandClient WaylandScannerTools)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(Wayland REQUIRED COMPONENTS Client)
find_package(WaylandScanner REQUIRED)
find_package(PlasmaWaylandProtocols REQUIRED)
include(GNUInstallDirs)
add_library(classic-raster STATIC ClassicPrimitives.cpp)
set_target_properties(classic-raster PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(classic-raster PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(classic-raster PUBLIC Qt6::Widgets)
add_library(win7classicqml2 MODULE qml/Plugin.cpp qml/ClassicItem.h qml/ClassicColors.h qml/TaskWindowIcon.cpp)
qt_generate_wayland_protocol_client_sources(win7classicqml2 PRIVATE_CODE FILES
    ${PLASMA_WAYLAND_PROTOCOLS_DIR}/plasma-window-management.xml)
target_link_libraries(win7classicqml2 PRIVATE classic-raster Qt6::Quick Qt6::Qml Qt6::WaylandClient Wayland::Client)
install(TARGETS win7classicqml2 DESTINATION ${CMAKE_INSTALL_DATADIR}/win7-classic/primitives2)
install(FILES qml/qmldir DESTINATION ${CMAKE_INSTALL_DATADIR}/win7-classic/primitives2)
add_library(win7classicstyle MODULE style/ClassicStyle.cpp)
target_link_libraries(win7classicstyle PRIVATE classic-raster)
# A live Qt application only searches its startup plugin paths; a user-local style cannot reach it.
if(CMAKE_INSTALL_PREFIX STREQUAL "/usr")
    install(TARGETS win7classicstyle DESTINATION lib/qt6/plugins/styles)
endif()
add_executable(win7-classic-probe Probe.cpp)
target_link_libraries(win7-classic-probe PRIVATE classic-raster)
install(TARGETS win7-classic-probe DESTINATION ${CMAKE_INSTALL_BINDIR})
