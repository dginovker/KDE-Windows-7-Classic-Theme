cmake_minimum_required(VERSION 3.20)
project(tray39-api-probe LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
include(CMakeFindDependencyMacro)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(KF6CoreAddons REQUIRED)
find_package(KF6PulseAudioQt REQUIRED)
find_package(KF6WindowSystem REQUIRED)
find_package(KF6IconThemes REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Qml)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Core DBus Quick Widgets)
find_package(Plasma REQUIRED)
find_package(PlasmaQuick REQUIRED)
set(PLASMA_SKIP_APPLET_INSTALL ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/plugins")
plasma_add_applet(org.win7classic.tray CPP_SOURCES tray.cpp status_notifiers.cpp status_notifiers.h notifier_menu.cpp notifier_menu.h QML_SOURCES main.qml RESOURCES main.xml show-hidden.png speaker-unmuted.png speaker-muted.png ethernet-disconnected.png)
target_link_libraries(org.win7classic.tray PRIVATE KF6::CoreAddons KF6::PulseAudioQt KF6::WindowSystem KF6::IconThemes Qt6::DBus Qt6::Quick Qt6::Widgets Plasma::Plasma)
add_executable(sni-probe sni_probe.cpp)
target_link_libraries(sni-probe PRIVATE Qt6::DBus)
