project(explorer-frequent-test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Core DBus)
find_package(KF6Config 6.31 REQUIRED)
set(classic ${CMAKE_CURRENT_SOURCE_DIR}/../../classic)
add_executable(win7-explorer-policy ${classic}/ExplorerPrivacyReader.cpp)
target_link_libraries(win7-explorer-policy PRIVATE Qt6::Core KF6::ConfigCore)

target_compile_definitions(probe PRIVATE WIN7_EXPLORER_POLICY="${CMAKE_CURRENT_SOURCE_DIR}/observe-reader.py")
